        function handleLogin(e) {
            e.preventDefault();
            const u = document.getElementById('login-username').value.trim();
            const p = document.getElementById('login-password').value.trim();

            if (u === 'hussein@sad1' && p === 'sad1981sad00') {
                currentUser = { id: 'admin_1', name: 'المدير حسين', username: u, role: 'admin' };
                localStorage.setItem('current_user', JSON.stringify(currentUser));
                showDashboard();
                return;
            }

            const agency = db.agencies.find(a => (a.username === u || a.phone === u) && a.password === p);
            if (agency) {
                if (new Date(agency.expiryDate) < new Date()) {
                    Swal.fire('خطأ', 'حساب الوكيل منتهي الصلاحية من قبل المدير!', 'error');
                    return;
                }
                currentUser = { id: agency.id, name: agency.name, username: agency.username, role: 'agency', phone: agency.phone };
                localStorage.setItem('current_user', JSON.stringify(currentUser));
                showDashboard();
                return;
            }

            const emp = db.employees.find(e => e.username === u && e.password === p);
            if (emp) {
                const ag = db.agencies.find(a => a.id === emp.agencyId);
                if (ag && new Date(ag.expiryDate) < new Date()) {
                    Swal.fire('خطأ', 'حساب الوكيل التابع له الموظف منتهي الصلاحية!', 'error');
                    return;
                }
                emp.lastDevice = navigator.userAgent;
                saveDb();

                currentUser = { id: emp.id, name: emp.name, username: emp.username, role: 'employee', agencyId: emp.agencyId };
                localStorage.setItem('current_user', JSON.stringify(currentUser));
                showDashboard();
                return;
            }

            Swal.fire('خطأ', 'اسم المستخدم أو كلمة المرور غير صحيحة!', 'error');
        }

        function extendAgency(id) {
            Swal.fire({
                title: 'تمديد صلاحية الوكيل',
                input: 'date',
                showCancelButton: true,
                confirmButtonText: 'تحديث',
                cancelButtonText: 'إلغاء'
            }).then((result) => {
                if (result.value) {
                    const ag = db.agencies.find(a => a.id === id);
                    if (ag) {
                        ag.expiryDate = result.value;
                        saveDb();
                        Swal.fire('تم', 'تم تحديث تاريخ انتهاء الصلاحية', 'success');
                        renderAdminDashboard();
                    }
                }
            });
        }

        function deleteAgency(id) {
            Swal.fire({
                title: 'هل أنت متأكد؟',
                text: 'سيتم حذف الوكيل وجميع بياناته!',
                icon: 'warning',
                showCancelButton: true,
                confirmButtonText: 'نعم، حذف',
                cancelButtonText: 'إلغاء'
            }).then((res) => {
                if (res.isConfirmed) {
                    db.agencies = db.agencies.filter(a => a.id !== id);
                    db.employees = db.employees.filter(e => e.agencyId !== id);
                    saveDb();
                    renderAdminDashboard();
                }
            });
        }

        function recordAttendance() {
            if (!navigator.geolocation) {
                Swal.fire('خطأ', 'خاصية تحديد الموقع غير مدعومة في متصفحك', 'error');
                return;
            }

            navigator.geolocation.getCurrentPosition((position) => {
                const userLat = position.coords.latitude;
                const userLng = position.coords.longitude;
                const emp = db.employees.find(e => e.id === currentUser.id);

                const distance = getDistanceFromLatLonInMeters(userLat, userLng, emp.lat, emp.lng);
                const allowedRadius = emp.radius || 100;

                if (distance > allowedRadius) {
                    Swal.fire('خارج النطاق', `أنت بعيد عن موقع المكتب المسجل بمسافة ${Math.round(distance)} متر. الحد المسموح ${allowedRadius} متر.`, 'error');
                    return;
                }

                const now = new Date();
                const currentTime = now.getHours().toString().padStart(2, '0') + ':' + now.getMinutes().toString().padStart(2, '0');
                const status = currentTime > emp.workStart ? 'متأخر' : 'حاضر';

                db.attendance.push({
                    id: 'att_' + Date.now(),
                    employeeId: emp.id,
                    agencyId: emp.agencyId,
                    date: now.toLocaleDateString(),
                    time: currentTime,
                    status: status
                });
                saveDb();
                Swal.fire('تم بنجاح', `تم تسجيل البصمة (${status}) بنجاح في الموقع المحدد`, 'success');
            }, () => {
                Swal.fire('خطأ', 'فشل في تحديد موقعك الجغرافي. تأكد من تفعيل الـ GPS.', 'error');
            });
        }

        function getDistanceFromLatLonInMeters(lat1, lon1, lat2, lon2) {
            const R = 6371000;
            const dLat = (lat2 - lat1) * (Math.PI / 180);
            const dLon = (lon2 - lon1) * (Math.PI / 180);
            const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * (Math.PI / 180)) * Math.cos(lat2 * (Math.PI / 180)) * Math.sin(dLon / 2) * Math.sin(dLon / 2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
            return R * c;
        }

        function openCompleteTaskModal(taskId) {
            Swal.fire({
                title: 'إنهاء المهمة وتقديم التفاصيل',
                html: `
                    <input id="swal-amount" type="number" class="swal2-input" placeholder="المبلغ أو الصرفيات (د.ع)">
                    <textarea id="swal-details" class="swal2-textarea" placeholder="تفاصيل الإنجاز والصرفيات..."></textarea>
                `,
                confirmButtonText: 'إرسال للإدارة',
                focusConfirm: false,
                preConfirm: () => {
                    return {
                        amount: document.getElementById('swal-amount').value,
                        details: document.getElementById('swal-details').value
                    }
                }
            }).then((result) => {
                if (result.isConfirmed) {
                    const task = db.tasks.find(t => t.id === taskId);
                    if (task) {
                        task.status = 'منجزة';
                        task.amount = parseFloat(result.value.amount) || 0;
                        task.details = result.value.details;
                        saveDb();
                        Swal.fire('تم', 'تم إرسال تفاصيل إنجاز المهمة بنجاح', 'success');
                        renderEmployeeDashboard();
                    }
                }
            });
        }
